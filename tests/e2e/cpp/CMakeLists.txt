cmake_minimum_required(VERSION 3.14)
project(nim_ffi_cpp_e2e CXX)

# Test harness compiles at C++20 so we can use designated initializers
# (`.field = …`) in test bodies; MSVC rejects them under /std:c++17.
# The generated example bindings and the CMake template they ship with
# stay at C++17, so library consumers aren't forced onto a newer std.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include("${CMAKE_CURRENT_SOURCE_DIR}/../sanitizer.cmake")

# Thread the sanitizer into every Nim dylib built below. `CACHE ... FORCE` is
# required (see the note in nim_ffi_lib.cmake).
set(NIM_FFI_EXTRA_ARGS ${NIM_SAN_ARGS}
    CACHE STRING "Extra nim c args when building the dylib" FORCE)

# Directory scope, not per target: gtest, tinycbor and the bindings' own example
# clients all link against the instrumented dylib and need the same flags.
if(SAN_CFLAGS)
    add_compile_options(${SAN_CFLAGS})
    add_link_options(${SAN_CFLAGS})
endif()

get_filename_component(_timer_cpp_bindings_dir
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../examples/timer/cpp_bindings"

    "${CMAKE_CURRENT_SOURCE_DIR}/../../../examples/echo/cpp_bindings"

include(GoogleTest)

# One gtest exe linked against the Nim libraries LIBS (CMake targets, each with a `<lib>_headers` and `<lib>_nim_lib`).
function(add_e2e_test target source)
    add_executable(${target} ${source})
    set(_rpath "")
    set(_stage "")
    foreach(_lib ${ARGN})
        target_link_libraries(${target} PRIVATE ${_lib}_headers)
        add_dependencies(${target} ${_lib}_nim_lib)
        get_target_property(_loc ${_lib} IMPORTED_LOCATION)
        get_filename_component(_dir "${_loc}" DIRECTORY)
        list(APPEND _rpath "${_dir}")
        list(APPEND _stage COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${${_lib}_RUNTIME_LIB}" "$<TARGET_FILE_DIR:${target}>")
    endforeach()
    target_link_libraries(${target} PRIVATE GTest::gtest_main)

    # Nim-built dylibs use `@rpath/lib*.dylib` install_names, so embed each library dir as an rpath; Windows has no rpath, so stage the DLLs next to the exe.
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Windows")
        set_target_properties(${target} PROPERTIES BUILD_RPATH "${_rpath}" INSTALL_RPATH "${_rpath}")
    else()
        add_custom_command(TARGET ${target} POST_BUILD ${_stage})
    endif()

    # Discover at test time: a POST_BUILD discovery launches the exe while Windows Defender still scans it and its DLLs, which overran the 5s timeout on CI.
    set(_props "")
    if(_san_test_env)
        set(_props PROPERTIES ENVIRONMENT "${_san_test_env}")
    endif()
    gtest_discover_tests(${target} DISCOVERY_MODE PRE_TEST DISCOVERY_TIMEOUT 120 ${_props})
endfunction()

add_e2e_test(timer_e2e_tests test_timer_e2e.cpp my_timer echo)
# Its own process: a stalled destroy quarantines a slot and leaves its loop firing.
add_e2e_test(ticker_e2e_tests test_ticker_e2e.cpp ticker)
